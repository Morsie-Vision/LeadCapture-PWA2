// ============================================================ // api-proxy.js - Sicheres Backend für DeepSeek API // ============================================================ const express = require('express'); const cors = require('cors'); const axios = require('axios'); const app = express(); const port = process.env.PORT || 3000; // ============================================================ // 1. SICHERE KONFIGURATION // ============================================================ // Diese Umgebungsvariablen werden auf Render gesetzt! const DEEPSEEK_API_KEY = process.env.DEEPSEEK_API_KEY; const PROXY_AUTH_TOKEN = process.env.PROXY_AUTH_TOKEN; // Prüfe, ob alle Keys gesetzt sind if (!DEEPSEEK_API_KEY) { console.error('❌ DEEPSEEK_API_KEY nicht gesetzt!'); console.error(' Bitte setze die Umgebungsvariable DEEPSEEK_API_KEY'); process.exit(1); } if (!PROXY_AUTH_TOKEN) { console.error('❌ PROXY_AUTH_TOKEN nicht gesetzt!'); console.error(' Bitte setze die Umgebungsvariable PROXY_AUTH_TOKEN'); process.exit(1); } const DEEPSEEK_API_URL = 'https://api.deepseek.com/v1/chat/completions'; console.log('✅ Konfiguration geladen:'); console.log(` DeepSeek Key: ${DEEPSEEK_API_KEY.substring(0, 10)}...`); console.log(` Auth Token: ${PROXY_AUTH_TOKEN.substring(0, 10)}...`); // ============================================================ // 2. MIDDLEWARE // ============================================================ // Erweiterte CORS-Einstellungen für Sicherheit const allowedOrigins = [ 'https://morsie-vision.github.io', // Deine GitHub Pages URL 'http://localhost:3000', // Lokale Entwicklung 'http://localhost:5500', // Live Server /\.github\.io$/ // Alle GitHub Pages Subdomains (Regex) ]; app.use(cors({ origin: function (origin, callback) { // Erlaube Anfragen ohne Origin (z.B. von mobilen Apps) if (!origin) return callback(null, true); // Prüfe, ob die Origin erlaubt ist const allowed = allowedOrigins.some(allowedOrigin => { if (typeof allowedOrigin === 'string') { return origin === allowedOrigin; } else if (allowedOrigin instanceof RegExp) { return allowedOrigin.test(origin); } return false; }); if (allowed) { callback(null, true); } else { console.warn(`⚠️ CORS-Block: ${origin} ist nicht erlaubt`); callback(new Error('Nicht erlaubt durch CORS')); } }, methods: ['GET', 'POST', 'HEAD', 'OPTIONS'], allowedHeaders: ['Content-Type', 'Authorization'] })); app.use(express.json({ limit: '50mb' })); app.use(express.urlencoded({ extended: true, limit: '50mb' })); // Logging-Middleware app.use((req, res, next) => { console.log(`📨 ${req.method} ${req.path} - ${req.ip}`); next(); }); // ============================================================ // 3. AUTHENTIFIZIERUNGS-MIDDLEWARE // ============================================================ function authenticate(req, res, next) { // Prüfe den Authorization-Header const authHeader = req.headers.authorization; if (!authHeader) { console.warn('⚠️ Kein Authorization-Header'); return res.status(401).json({ success: false, error: '❌ Kein Authorization-Header gefunden' }); } // Erwartet Format: "Bearer " const parts = authHeader.split(' '); if (parts.length !== 2 || parts[0] !== 'Bearer') { console.warn('⚠️ Ungültiges Authorization-Format:', authHeader); return res.status(401).json({ success: false, error: '❌ Ungültiges Authorization-Format. Erwartet: Bearer TOKEN' }); } const token = parts[1]; if (!token || token !== PROXY_AUTH_TOKEN) { console.warn('⚠️ Ungültiger Token'); return res.status(403).json({ success: false, error: '❌ Ungültiger Authentifizierungs-Token' }); } console.log('✅ Authentifizierung erfolgreich'); // Token ist gültig -> weiter zur eigentlichen Logik next(); } // ============================================================ // 4. HEALTH CHECK (OHNE Authentifizierung) // ============================================================ app.head('/api/process-image', (req, res) => { res.status(200).send('OK'); }); app.get('/api/health', (req, res) => { res.json({ status: 'OK', timestamp: new Date().toISOString(), environment: process.env.NODE_ENV || 'production' }); }); // ============================================================ // 5. HAUPT-ENDPUNKT (MIT Authentifizierung) // ============================================================ app.post('/api/process-image', authenticate, async (req, res) => { const startTime = Date.now(); try { const { image } = req.body; if (!image) { return res.status(400).json({ success: false, error: 'Kein Bild übergeben' }); } console.log('📸 Bild empfangen, Größe:', (image.length / 1024).toFixed(2), 'KB'); // System-Prompt für DeepSeek const systemPrompt = ` Du bist ein OCR- und Parsing-Experte für Messeleads. Extrahiere die folgenden Felder aus dem bereitgestellten Bild eines Fragebogens oder einer Visitenkarte: - Vorname, Nachname, Firma, Position, E-Mail, Telefon, Adresse (Straße, PLZ, Stadt), Interessensgebiete (Liste), Besuchertyp. Gib das Ergebnis **ausschließlich** als gültiges JSON-Objekt zurück, das genau dieser Struktur entspricht: { "firstName": "", "lastName": "", "company": "", "position": "", "email": "", "phone": "", "address": { "street": "", "zip": "", "city": "" }, "interests": [], "visitorType": "" } Verwende leere Zeichenketten oder leere Arrays, wenn Informationen fehlen. Antworte NUR mit dem JSON-Objekt, ohne zusätzlichen Text. `; const requestBody = { model: 'deepseek-v4-flash', messages: [ { role: 'system', content: systemPrompt }, { role: 'user', content: [ { type: 'text', text: 'Extrahiere die Lead-Informationen aus diesem Bild:' }, { type: 'image_url', image_url: { url: image } } ] } ], temperature: 0.1, response_format: { type: 'json_object' } }; console.log('🚀 Sende Anfrage an DeepSeek API...'); const response = await axios.post(DEEPSEEK_API_URL, requestBody, { headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${DEEPSEEK_API_KEY}` }, timeout: 30000 // 30 Sekunden Timeout }); // Antwort parsen const content = response.data.choices[0]?.message?.content; if (!content) { throw new Error('Leere Antwort von DeepSeek erhalten.'); } const leadData = JSON.parse(content); const duration = Date.now() - startTime; console.log(`✅ Lead erfolgreich geparst (${duration}ms)`); // Erfolgreiche Antwort res.json({ success: true, data: leadData, processingTime: duration }); } catch (error) { const duration = Date.now() - startTime; console.error(`❌ Fehler (${duration}ms):`, error.message); // Detaillierte Fehlerbehandlung let errorMessage = 'Interner Serverfehler'; let statusCode = 500; if (error.response) { // Die Anfrage wurde gesendet und der Server hat geantwortet console.error('📨 DeepSeek API Response:', error.response.status); if (error.response.data) { console.error('📨 Response Data:', JSON.stringify(error.response.data, null, 2)); } errorMessage = error.response.data?.error?.message || `DeepSeek API Fehler (${error.response.status})`; statusCode = error.response.status; } else if (error.request) { // Die Anfrage wurde gesendet, aber keine Antwort erhalten console.error('⏰ Keine Antwort von DeepSeek API'); errorMessage = 'DeepSeek API antwortet nicht (Timeout)'; statusCode = 504; } else { // Etwas anderes ist schiefgelaufen errorMessage = error.message; statusCode = 500; } res.status(statusCode).json({ success: false, error: errorMessage, processingTime: duration }); } }); // ============================================================ // 6. 404 HANDLER // ============================================================ app.use((req, res) => { res.status(404).json({ success: false, error: 'Endpunkt nicht gefunden' }); }); // ============================================================ // 7. FEHLERBEHANDLUNG FÜR UNERWARTETE FEHLER // ============================================================ app.use((err, req, res, next) => { console.error('💥 Unerwarteter Fehler:', err); res.status(500).json({ success: false, error: 'Interner Serverfehler' }); }); // ============================================================ // 8. SERVER STARTEN // ============================================================ app.listen(port, '0.0.0.0', () => { console.log('\n' + '='.repeat(60)); console.log('✅ DeepSeek Proxy Server läuft erfolgreich!'); console.log('='.repeat(60)); console.log(`📡 Port: ${port}`); console.log(`🔑 DeepSeek Key: ${DEEPSEEK_API_KEY ? '✅ GESETZT' : '❌ FEHLT'}`); console.log(`🔐 Auth Token: ${PROXY_AUTH_TOKEN ? '✅ GESETZT' : '❌ FEHLT'}`); console.log(`🌐 Endpunkt: http://localhost:${port}/api/process-image`); console.log('='.repeat(60) + '\n'); }); // Uncaught Exception Handler process.on('uncaughtException', (error) => { console.error('💥 Uncaught Exception:', error); }); // Unhandled Rejection Handler process.on('unhandledRejection', (reason, promise) => { console.error('💥 Unhandled Rejection:', reason); });
